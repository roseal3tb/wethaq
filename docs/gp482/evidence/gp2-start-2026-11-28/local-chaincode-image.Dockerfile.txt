# Local test image only: same runtime stage as chaincode/shipment/Dockerfile, with dependencies installed on the
# host, because npm inside Docker cannot reach the registry in this sandbox (blocker B11).
FROM node:22-bookworm-slim
WORKDIR /app
COPY package.json ./
COPY node_modules ./node_modules
COPY dist ./dist
ENV CHAINCODE_SERVER_ADDRESS=0.0.0.0:9999
EXPOSE 9999
USER node
CMD ["sh", "-c", "exec node_modules/.bin/fabric-chaincode-node server --chaincode-address=$CHAINCODE_SERVER_ADDRESS --chaincode-id=$CHAINCODE_ID"]
