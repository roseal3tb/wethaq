import { test } from '@playwright/test';
import { mkdirSync } from 'node:fs';
const OUT = process.env.A11Y_OUT || '/tmp/shot';
for (const width of [390, 768, 1440]) for (const theme of ['light', 'dark']) for (const path of ['/overview', '/forecasts', '/inventory', '/shipments', '/users']) {
  test(`shot ${width} ${theme} ${path}`, async ({ page }) => {
    await page.setViewportSize({ width, height: 900 });
    await page.emulateMedia({ colorScheme: theme as 'light' | 'dark', reducedMotion: 'reduce' });
    await page.goto(path);
    if (page.url().endsWith('/login')) { await page.getByRole('button', { name: 'Sign in to Wethaq' }).click(); await page.waitForURL(u => !u.pathname.endsWith('/login')); }
    if (path === '/users') { await page.getByRole('button', { name: /Demo role:/ }).click(); await page.getByRole('menuitemradio', { name: 'Admin', exact: true }).click(); await page.goto(path); }
    await page.waitForLoadState('networkidle'); await page.waitForTimeout(1500);
    mkdirSync(OUT, { recursive: true });
    await page.screenshot({ path: `${OUT}/${width}-${theme}${path.replace(/\//g, '_')}.png`, fullPage: true, animations: 'disabled' });
  });
}
