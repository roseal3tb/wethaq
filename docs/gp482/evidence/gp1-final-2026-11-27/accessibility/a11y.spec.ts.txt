import { test } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';
import { writeFileSync, mkdirSync } from 'node:fs';

const pages = ['/login', '/overview', '/forecasts', '/inventory', '/shipments', '/users', '/does-not-exist'];
const all: Record<string, unknown> = {};

for (const theme of ['light', 'dark']) {
  test(`axe ${theme}`, async ({ page }) => {
    test.setTimeout(300_000);
    await page.emulateMedia({ colorScheme: theme as 'light' | 'dark' });
    for (const path of pages) {
      if (path !== '/login') {
        await page.goto(path);
        if (page.url().endsWith('/login')) { await page.getByRole('button', { name: 'Sign in to Wethaq' }).click(); await page.waitForURL(u => !u.pathname.endsWith('/login')); }
      } else { await page.goto(path); }
      await page.waitForLoadState('networkidle');
      await page.waitForTimeout(800);
      const r = await new AxeBuilder({ page }).withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa', 'best-practice']).analyze();
      all[`${theme} ${path}`] = r.violations.map(v => ({ id: v.id, impact: v.impact, help: v.help, tags: v.tags.filter(t => t.startsWith('wcag')), nodes: v.nodes.length, targets: v.nodes.slice(0, 6).map(n => ({ target: n.target.join(' '), summary: n.failureSummary?.split('\n').slice(0, 3).join(' | '), html: n.html.slice(0, 160) })) }));
      if (path === '/login' && theme === 'light') { /* stay signed out for login */ }
    }
    mkdirSync('/tmp/a11y', { recursive: true });
    writeFileSync(`/tmp/a11y/axe-${theme}.json`, JSON.stringify(all, null, 2));
  });
}
